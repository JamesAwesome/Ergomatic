```ts
test("the log detail reads the Concept2 link exactly ONCE, on a ready row", async ({
  page,
}) => {
  // TD-4's gate. Written and run BEFORE the fix (RF35): a mutation of the
  // FIXED code would test the fix's shape rather than the defect, so this
  // must be seen to read 2 on the unmodified tree first.
  //
  // TWO STEPS, not one poll. `expect.poll(...).toBe(readsBefore + 1)`
  // evaluates on entry and can sample between the two increments, so on the
  // unfixed build it passes by luck roughly as often as it fails — this
  // file's own finding at the `MANUAL:` test above. The poll below is the
  // readiness precondition; the bare `expect` after the block has rendered
  // is the gate.
  const fake = await signIn(page, "one-link-read");
  fake.linked();
  await postMonitorLog(page, "Single Read");

  const readsBefore = fake.linkReads;
  await openLogDetail(page, "Single Read");
  await expect.poll(() => fake.linkReads).toBeGreaterThan(readsBefore);
  // The send block rendering is the work that would surface a duplicate: it
  // is the second consumer, and it cannot draw until its own read resolved.
  await expect(page.locator(".c2-send")).toBeVisible();
  expect(fake.linkReads).toBe(readsBefore + 1);
});
```

```ts
  it("a reconciliation whose store write THROWS still answers 200, and still records the send", async () => {
    // The other half of the catch at `routes/concept2.ts:1239-1266`. Its
    // sibling — that a working reconciliation reaches the DB — is gated
    // above; this is the failure path, which four earlier attempts could not
    // gate because every one produced a 500 from the FIXTURE rather than
    // from the code under test.
    //
    // THE FAILURE IS ROW-SCOPED, and that is the whole trick. Dropping or
    // revoking `session_logs.verified` cannot work: `recordC2Result` writes
    // the same column on the same table and `logs.get` selects it on the
    // handler's first statement, so the request breaks instead of the
    // reconciliation. A trigger keyed to ONE row id fails
    // `markC2Verified`'s UPDATE and nothing else.
    const { bearer, userId } = await signIn("send-reconcile-throws");
    await linkAccount({
      userId,
      c2UserId: C2_USER_RECONCILE_THROWS,
      username: "jamesawesome",
    });

    const olderId = await postLog(bearer);
    fetchMock.mockImplementation(async (input: Parameters<typeof fetch>[0]) => {
      const url = String(input);
      if (url.includes("/api/users/me/results?")) {
        return jsonResponse(200, declarationListBody("H"));
      }
      return jsonResponse(201, created201(9201, C2_USER_RECONCILE_THROWS));
    });
    await request(app)
      .post(`/api/concept2/results/${olderId}`)
      .set("Authorization", bearer)
      .send({ tz: "Europe/London" });
    expect((await readRow(bearer, olderId)).verified).not.toBe(true);

    // Interpolated, not parameterised: CREATE TRIGGER takes no bind
    // parameters. `olderId` is a UUID this test just minted through the
    // real route, so it is ours and no sibling test can hold it.
    await pool.query(`
      create or replace function td1_block_verify() returns trigger
      language plpgsql as $$
      begin raise exception 'TD-1: forced markC2Verified failure'; end $$;
    `);
    await pool.query(`
      create trigger td1_block_verify_trg before update on session_logs
      for each row when (old.id = '${olderId}'::uuid)
      execute function td1_block_verify();
    `);

    try {
      const freshId = await postLog(bearer);
      fetchMock.mockImplementation(
        async (input: Parameters<typeof fetch>[0]) => {
          const url = String(input);
          if (url.includes("/api/users/me/results?")) {
            const body = declarationListBody("H");
            return jsonResponse(200, {
              data: [
                {
                  id: 9201,
                  type: "rower",
                  weight_class: "H",
                  date_utc: "2026-08-22 10:00:30",
                  date: "2026-08-22 06:00:30",
                  verified: true,
                },
                ...body.data,
              ],
            });
          }
          return jsonResponse(201, created201(9202, C2_USER_RECONCILE_THROWS));
        },
      );

      const second = await request(app)
        .post(`/api/concept2/results/${freshId}`)
        .set("Authorization", bearer)
        .send({ tz: "Europe/London" });

      // THE INVARIANT: the send is unaffected.
      expect(second.status).toBe(200);
      // And it actually recorded — a 200 that wrote nothing would satisfy
      // the status assertion while losing the send.
      expect((await readRow(bearer, freshId)).c2ResultId).toBe(9202);
      // The older row did NOT upgrade, which is what proves the
      // reconciliation genuinely failed rather than being skipped.
      expect((await readRow(bearer, olderId)).verified).not.toBe(true);
    } finally {
      // The container is shared by every test in this file.
      await pool.query(
        "drop trigger if exists td1_block_verify_trg on session_logs",
      );
      await pool.query("drop function if exists td1_block_verify()");
    }
  });
```

```ts
test("justrow-log", async ({ page }) => {
  await openJustRowLive(page, "screenshots-justrow-log@e2e.test");
  await page.getByRole("button", { name: "End session" }).click();
  await page.getByRole("button", { name: "Tap again to end" }).click();

  // STILL ON THE FREE-ROW ROUTE. This is the free-row analogue of
  // `connected.spec.ts:919`, and it replaces the "Wrapping up" check the
  // 2026-09-07 attempt used — `ConnectedSurface.tsx:466-472` says that text
  // renders on EVERY ended state, held or not, so it proved nothing.
  await expect(page).toHaveURL(/\/justrow$/);

  // The window: after the terminated frame, before BURST_LINGER_MS (2000ms)
  // closes. Deliver too early and the driver refuses it `out-of-window`;
  // too late and it logs `terminate-observations` and files NOTHING (spec
  // §1.3, ORDER E).
  //
  // TOTALS ARE WHAT THIS SESSION ACTUALLY ROWED, not the script's full
  // length. `openJustRowLive` gates on frame 16 (`0:16`, 76 m) and End is
  // pressed straight after, so 16 s / 76 m is the honest pair — the same
  // convention `:5818-5823` follows for the programmed capture ("this
  // fixture's own rowed numbers"). The script runs to 90 frames, but those
  // never happen here: seeding 90 s / 520 m would put a total on screen the
  // row did not produce, and at 500x90/520 it would also read 1:26.5 —
  // faster than any human has rowed. 500 x 16 / 76 = 105.3 s = 1:45.3,
  // which is the figure this fixture's own header comment computes.
  await page.evaluate(() => {
    window.__pm5FakeControls__?.deliverSummary({
      elapsedSeconds: 16,
      meters: 76,
    });
  });

  await expect(page).toHaveURL(/\/justrow\/log$/, { timeout: 15_000 });
  await expect(page.getByText("EFFORT", { exact: true })).toBeVisible();
  // THE GATE, and it is a locator rather than the PNG: nothing can go red
  // on an image. Three tiles carry figures and three dash (spec §4.3).
  const tier = page.getByTestId("summary-machine-tier");
  await expect(tier).toBeVisible();
  await expect(tier.getByRole("group", { name: "AVG WATTS" })).toContainText(
    /\d/,
  );
  await expect(tier.getByRole("group", { name: "DRAG" })).toContainText("128");
  await expect(tier.getByRole("group", { name: "RATE" })).toContainText("24");

  await page.screenshot({
    path: path.join(SCREENSHOTS_DIR, "justrow-log.png"),
  });
});
```
